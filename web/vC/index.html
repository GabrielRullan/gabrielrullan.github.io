def audit_transaction_stream(df_records, tolerance_eur=0.01):
    """
    Forensic reconciliation between billing ledger
    and statutory ERP journal entries.
    """
    discrepancies = df_records[
        (df_records['erp_ledger_amt'] - df_records['billing_amt']).abs() > tolerance_eur
    ]
    if not discrepancies.empty:
        log_compliance_alert(
            count=len(discrepancies),
            total_delta=discrepancies['delta'].sum(),
            severity='HIGH_RISK_AUDIT_TRIGGER'
        )
    return discrepancies
